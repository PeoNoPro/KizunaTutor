-- ====================================================================================================
-- 3. HÀM TRIGGER CẬP NHẬT THỜI GIAN (updated_at)
-- ====================================================================================================

CREATE OR REPLACE FUNCTION update_timestamp_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ====================================================================================================
-- PHÂN HỆ 1: NGƯỜI DÙNG & XÁC THỰC PHÂN QUYỀN (USERS & ROLES)
-- ====================================================================================================

-- 1. Bảng Vai Trò (Roles)
DROP TABLE IF EXISTS roles CASCADE;
CREATE TABLE roles (
    role_id SMALLSERIAL PRIMARY KEY,
    role_name VARCHAR(50) NOT NULL UNIQUE,
    description VARCHAR(255),
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

-- 2. Bảng Người Dùng (Users)
DROP TABLE IF EXISTS users CASCADE;
CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role_id SMALLINT NOT NULL REFERENCES roles(role_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    avatar_url VARCHAR(500),
    account_status user_account_status_enum NOT NULL DEFAULT 'ACTIVE',
    must_change_password BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ DEFAULT NULL
);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_role_id ON users(role_id);
CREATE INDEX idx_users_status ON users(account_status);
CREATE TRIGGER trg_users_updated_at BEFORE UPDATE ON users FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 3. Bảng Hồ Sơ Học Viên (Learners)
DROP TABLE IF EXISTS learners CASCADE;
CREATE TABLE learners (
    learner_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL UNIQUE REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    current_level VARCHAR(10),
    target_level VARCHAR(10),
    learning_goals TEXT,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE TRIGGER trg_learners_updated_at BEFORE UPDATE ON learners FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- ====================================================================================================
-- PHÂN HỆ 2: HỒ SƠ GIA SƯ & THẨM ĐỊNH (TUTOR PROFILE & VERIFICATION)

-- 4. Bảng Hồ Sơ Gia Sư (Tutors)
DROP TABLE IF EXISTS tutors CASCADE;
CREATE TABLE tutors (
    tutor_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL UNIQUE REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    headline VARCHAR(200),
    bio TEXT,
    experience_years SMALLINT NOT NULL DEFAULT 0 CHECK (experience_years >= 0),
    hourly_rate NUMERIC(12, 2) NOT NULL DEFAULT 0.00,
    highest_jlpt_level VARCHAR(10),
    avg_rating NUMERIC(3, 2) NOT NULL DEFAULT 0.00 CHECK (avg_rating BETWEEN 0.00 AND 5.00),
    total_reviews INTEGER NOT NULL DEFAULT 0 CHECK (total_reviews >= 0),
    verification_status tutor_verification_status_enum NOT NULL DEFAULT 'UNVERIFIED',
    membership_status tutor_membership_status_enum NOT NULL DEFAULT 'FREE',
    is_featured BOOLEAN NOT NULL DEFAULT FALSE,
    tutor_state VARCHAR(30) NOT NULL DEFAULT 'DRAFT_PROFILE',
    phone VARCHAR(20),
    date_of_birth DATE,
    address VARCHAR(255),
    certificate_urls TEXT,
    verification_feedback TEXT,
    submitted_at TIMESTAMPTZ,
    verified_at TIMESTAMPTZ,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_tutors_verif_status ON tutors(verification_status);
CREATE INDEX idx_tutors_rating ON tutors(avg_rating);
CREATE INDEX idx_tutors_hourly_rate ON tutors(hourly_rate);
CREATE INDEX idx_tutors_is_featured ON tutors(is_featured);
CREATE TRIGGER trg_tutors_updated_at BEFORE UPDATE ON tutors FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 5. Bảng Chứng Chỉ / Bằng Cấp Gia Sư (Tutor Credentials)
DROP TABLE IF EXISTS tutor_credentials CASCADE;
CREATE TABLE tutor_credentials (
    credential_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    credential_type VARCHAR(50) NOT NULL,
    title VARCHAR(150) NOT NULL,
    file_url VARCHAR(500) NOT NULL,
    verified_status tutor_credential_status_enum NOT NULL DEFAULT 'PENDING',
    rejection_reason TEXT,
    uploaded_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_credentials_tutor_id ON tutor_credentials(tutor_id);
CREATE INDEX idx_credentials_status ON tutor_credentials(verified_status);

-- 6. Bảng Nhật Ký Thẩm Định Gia Sư (Tutor Verification Histories)
DROP TABLE IF EXISTS tutor_verification_histories CASCADE;
CREATE TABLE tutor_verification_histories (
    history_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    officer_id UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    action tutor_verif_action_enum NOT NULL,
    notes TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_verif_hist_tutor ON tutor_verification_histories(tutor_id);
CREATE INDEX idx_verif_hist_officer ON tutor_verification_histories(officer_id);

-- 7. Bảng Lịch Rảnh Giảng Dạy Của Gia Sư (Tutor Availabilities)
DROP TABLE IF EXISTS tutor_availabilities CASCADE;
CREATE TABLE tutor_availabilities (
    availability_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    day_of_week SMALLINT NOT NULL CHECK (day_of_week BETWEEN 1 AND 7),
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT chk_avail_time CHECK (start_time < end_time)
);
CREATE INDEX idx_avail_tutor_day ON tutor_availabilities(tutor_id, day_of_week);

-- 8. Bảng Yêu Cầu Kết Nối (Connection Requests)
DROP TABLE IF EXISTS connection_requests CASCADE;
CREATE TABLE connection_requests (
    request_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    initial_message TEXT,
    status connection_request_status_enum NOT NULL DEFAULT 'PENDING',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    responded_at TIMESTAMPTZ DEFAULT NULL
);
CREATE INDEX idx_conn_learner ON connection_requests(learner_id);
CREATE INDEX idx_conn_tutor ON connection_requests(tutor_id);
CREATE INDEX idx_conn_status ON connection_requests(status);

-- 9. Bảng Đặt Lịch Học Thử (Trial Bookings)
DROP TABLE IF EXISTS trial_bookings CASCADE;
CREATE TABLE trial_bookings (
    booking_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    request_id UUID REFERENCES connection_requests(request_id) ON DELETE SET NULL ON UPDATE CASCADE,
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    scheduled_start_time TIMESTAMPTZ NOT NULL,
    scheduled_end_time TIMESTAMPTZ NOT NULL,
    meeting_link VARCHAR(500),
    booking_status trial_booking_status_enum NOT NULL DEFAULT 'PENDING',
    learner_confirmed BOOLEAN NOT NULL DEFAULT FALSE,
    tutor_confirmed BOOLEAN NOT NULL DEFAULT FALSE,
    official_arrangement_agreed BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_trial_time CHECK (scheduled_start_time < scheduled_end_time)
);
CREATE INDEX idx_trial_learner ON trial_bookings(learner_id);
CREATE INDEX idx_trial_tutor ON trial_bookings(tutor_id);
CREATE INDEX idx_trial_status ON trial_bookings(booking_status);
CREATE INDEX idx_trial_time ON trial_bookings(scheduled_start_time, scheduled_end_time);
CREATE TRIGGER trg_trial_bookings_updated_at BEFORE UPDATE ON trial_bookings FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 10. Bảng Nhật Ký Thay Đổi Trạng Thái Học Thử (Trial Booking Histories)
DROP TABLE IF EXISTS trial_booking_histories CASCADE;
CREATE TABLE trial_booking_histories (
    history_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    booking_id UUID NOT NULL REFERENCES trial_bookings(booking_id) ON DELETE CASCADE ON UPDATE CASCADE,
    changed_by_user_id UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    previous_status trial_booking_status_enum,
    new_status trial_booking_status_enum NOT NULL,
    change_reason VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_trial_hist_booking ON trial_booking_histories(booking_id);

-- 11. Bảng Không Gian Học Tập Chính Thức (Learning Spaces)
DROP TABLE IF EXISTS learning_spaces CASCADE;
CREATE TABLE learning_spaces (
    space_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    trial_booking_id UUID REFERENCES trial_bookings(booking_id) ON DELETE SET NULL,
    title VARCHAR(150) NOT NULL,
    status learning_space_status_enum NOT NULL DEFAULT 'ACTIVE',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_spaces_tutor_learner ON learning_spaces(tutor_id, learner_id);
CREATE TRIGGER trg_spaces_updated_at BEFORE UPDATE ON learning_spaces FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 12. Bảng Tài Liệu Học Tập (Learning Materials)
DROP TABLE IF EXISTS learning_materials CASCADE;
CREATE TABLE learning_materials (
    material_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    space_id UUID NOT NULL REFERENCES learning_spaces(space_id) ON DELETE CASCADE ON UPDATE CASCADE,
    title VARCHAR(150) NOT NULL,
    file_url VARCHAR(500) NOT NULL,
    file_type VARCHAR(50),
    uploaded_by UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_materials_space ON learning_materials(space_id);

-- 13. Bảng Bài Tập Về Nhà (Assignments)
DROP TABLE IF EXISTS assignments CASCADE;
CREATE TABLE assignments (
    assignment_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    space_id UUID NOT NULL REFERENCES learning_spaces(space_id) ON DELETE CASCADE ON UPDATE CASCADE,
    title VARCHAR(150) NOT NULL,
    description TEXT,
    due_date TIMESTAMPTZ,
    total_points NUMERIC(5, 2) NOT NULL DEFAULT 100.00,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_assignments_space_id ON assignments(space_id);
CREATE TRIGGER trg_assignments_updated_at BEFORE UPDATE ON assignments FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 14. Bảng Bài Nộp Của Học Viên (Assignment Submissions)
DROP TABLE IF EXISTS assignment_submissions CASCADE;
CREATE TABLE assignment_submissions (
    submission_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    assignment_id UUID NOT NULL REFERENCES assignments(assignment_id) ON DELETE CASCADE ON UPDATE CASCADE,
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    attempt_number SMALLINT NOT NULL DEFAULT 1 CHECK (attempt_number > 0),
    submission_content TEXT,
    attachment_url VARCHAR(500),
    ai_score NUMERIC(5, 2),
    ai_feedback TEXT,
    grade NUMERIC(5, 2),
    tutor_feedback TEXT,
    is_finalized BOOLEAN NOT NULL DEFAULT FALSE,
    status assignment_submission_status_enum NOT NULL DEFAULT 'SUBMITTED',
    submitted_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    graded_at TIMESTAMPTZ DEFAULT NULL
);
CREATE INDEX idx_submissions_assign_learner ON assignment_submissions(assignment_id, learner_id);

-- 15. Bảng Kho Từ Điển Gốc Kanji (Kanji Dictionary)
DROP TABLE IF EXISTS kanji_dictionary CASCADE;
CREATE TABLE kanji_dictionary (
    kanji_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    kanji_character CHAR(1) NOT NULL UNIQUE,
    unicode_hex CHAR(5) NOT NULL UNIQUE,
    radical VARCHAR(4),
    stroke_count SMALLINT NOT NULL CHECK (stroke_count >= 0),
    han_viet VARCHAR(50),
    onyomi VARCHAR(50),
    kunyomi VARCHAR(100),
    meaning VARCHAR(200),
    jlpt_level jlpt_level_enum,
    svg_data TEXT,
    stroke_order_svg_url VARCHAR(500),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_kanji_char ON kanji_dictionary(kanji_character);
CREATE INDEX idx_kanji_jlpt ON kanji_dictionary(jlpt_level);
CREATE INDEX idx_kanji_han_viet ON kanji_dictionary(han_viet);
CREATE TRIGGER trg_kanji_updated_at BEFORE UPDATE ON kanji_dictionary FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 16. Bảng Mục Kanji Trong Bài Tập (Assignment Kanji Items)
DROP TABLE IF EXISTS assignment_kanji_items CASCADE;
CREATE TABLE assignment_kanji_items (
    item_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    assignment_id UUID NOT NULL REFERENCES assignments(assignment_id) ON DELETE CASCADE ON UPDATE CASCADE,
    kanji_id UUID NOT NULL REFERENCES kanji_dictionary(kanji_id) ON DELETE CASCADE ON UPDATE CASCADE,
    target_accuracy NUMERIC(4, 1) NOT NULL DEFAULT 80.0,
    min_practice_times SMALLINT NOT NULL DEFAULT 3 CHECK (min_practice_times >= 0),
    order_index INTEGER NOT NULL DEFAULT 0,
    CONSTRAINT uq_assign_kanji UNIQUE (assignment_id, kanji_id)
);

-- 17. Bảng Bài Thực Hành Luyện Chữ Hán Của Học Viên (Kanji Practice Submissions)
DROP TABLE IF EXISTS kanji_practice_submissions CASCADE;
CREATE TABLE kanji_practice_submissions (
    practice_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    assignment_item_id UUID REFERENCES assignment_kanji_items(item_id) ON DELETE SET NULL ON UPDATE CASCADE,
    kanji_id UUID REFERENCES kanji_dictionary(kanji_id) ON DELETE SET NULL ON UPDATE CASCADE,
    kanji_character CHAR(1) NOT NULL,
    canvas_vector_data JSONB,
    drawing_image_url VARCHAR(500),
    ai_accuracy_score NUMERIC(4, 1) CHECK (ai_accuracy_score BETWEEN 0.0 AND 100.0),
    ai_analysis_result JSONB,
    status practice_submission_status_enum NOT NULL DEFAULT 'PENDING',
    practiced_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_practice_learner ON kanji_practice_submissions(learner_id);
CREATE INDEX idx_practice_status ON kanji_practice_submissions(status);

-- 18. Bảng Ghi Nhận Tiến Độ Học Chữ Hán (Learning Progress Records)
DROP TABLE IF EXISTS learning_progress_records CASCADE;
CREATE TABLE learning_progress_records (
    progress_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    kanji_id UUID NOT NULL REFERENCES kanji_dictionary(kanji_id) ON DELETE CASCADE ON UPDATE CASCADE,
    total_practices INTEGER NOT NULL DEFAULT 0 CHECK (total_practices >= 0),
    highest_score NUMERIC(4, 1) NOT NULL DEFAULT 0.0,
    is_mastered BOOLEAN NOT NULL DEFAULT FALSE,
    last_practiced_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_learner_kanji_progress UNIQUE (learner_id, kanji_id)
);

-- 19. Bảng Thư Mục / Bộ Thẻ Kanji Cá Nhân Của Người Dùng (Kanji Folders)
DROP TABLE IF EXISTS kanji_folders CASCADE;
CREATE TABLE kanji_folders (
    folder_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    folder_name VARCHAR(100) NOT NULL,
    description VARCHAR(500),
    color VARCHAR(30) DEFAULT '#EF4444',
    is_default BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_kanji_folders_user ON kanji_folders(user_id);
CREATE TRIGGER trg_kanji_folders_updated_at BEFORE UPDATE ON kanji_folders FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 20. Bảng Liên Kết Chữ Kanji Thuộc Bộ Thẻ (Kanji Folder Items)
DROP TABLE IF EXISTS kanji_folder_items CASCADE;
CREATE TABLE kanji_folder_items (
    item_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    folder_id UUID NOT NULL REFERENCES kanji_folders(folder_id) ON DELETE CASCADE ON UPDATE CASCADE,
    kanji_id UUID NOT NULL REFERENCES kanji_dictionary(kanji_id) ON DELETE CASCADE ON UPDATE CASCADE,
    order_index INTEGER NOT NULL DEFAULT 0,
    notes VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_folder_kanji UNIQUE (folder_id, kanji_id)
);
CREATE INDEX idx_folder_items_folder ON kanji_folder_items(folder_id);
CREATE INDEX idx_folder_items_kanji ON kanji_folder_items(kanji_id);

-- 21. Bảng Phiên Chơi Minigame Bắn Chữ Kanji (Minigame Sessions)
DROP TABLE IF EXISTS minigame_sessions CASCADE;
CREATE TABLE minigame_sessions (
    session_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    folder_id UUID REFERENCES kanji_folders(folder_id) ON DELETE SET NULL ON UPDATE CASCADE,
    folder_name VARCHAR(100),
    duration_seconds INTEGER NOT NULL,
    target_duration_seconds INTEGER,
    words_typed_count INTEGER NOT NULL,
    total_mistakes INTEGER NOT NULL DEFAULT 0,
    accuracy NUMERIC(5, 2) NOT NULL,
    score INTEGER NOT NULL DEFAULT 0,
    is_won BOOLEAN NOT NULL,
    failed_kanji_id UUID REFERENCES kanji_dictionary(kanji_id) ON DELETE SET NULL ON UPDATE CASCADE,
    failed_kanji_character VARCHAR(10),
    played_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_minigame_sessions_user ON minigame_sessions(user_id);
CREATE INDEX idx_minigame_sessions_played ON minigame_sessions(played_at DESC);

-- 22. Bảng Ghi Nhận Lỗi Gõ Sai Trong Minigame (Minigame Mistake Logs)
DROP TABLE IF EXISTS minigame_mistake_logs CASCADE;
CREATE TABLE minigame_mistake_logs (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES minigame_sessions(session_id) ON DELETE CASCADE ON UPDATE CASCADE,
    kanji_id UUID REFERENCES kanji_dictionary(kanji_id) ON DELETE SET NULL ON UPDATE CASCADE,
    kanji_character VARCHAR(10) NOT NULL,
    reading VARCHAR(50),
    mistake_count INTEGER NOT NULL DEFAULT 1,
    typed_input VARCHAR(100),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_minigame_mistakes_session ON minigame_mistake_logs(session_id);

-- ====================================================================================================
-- PHÂN HỆ 5: ĐÁNH GIÁ, GÓI THÀNH VIÊN & THANH TOÁN (REVIEWS, MEMBERSHIP & PAYMENTS)
-- ====================================================================================================

-- 23. Bảng Đánh Giá Gia Sư (Tutor Reviews)
DROP TABLE IF EXISTS tutor_reviews CASCADE;
CREATE TABLE tutor_reviews (
    review_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    learner_id UUID NOT NULL REFERENCES learners(learner_id) ON DELETE CASCADE ON UPDATE CASCADE,
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    space_id UUID REFERENCES learning_spaces(space_id) ON DELETE SET NULL ON UPDATE CASCADE,
    rating NUMERIC(2, 1) NOT NULL CHECK (rating BETWEEN 1.0 AND 5.0),
    comment TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_reviews_tutor_id ON tutor_reviews(tutor_id);

-- 24. Bảng Gói Thành Viên Gia Sư (Membership Plans)
DROP TABLE IF EXISTS membership_plans CASCADE;
CREATE TABLE membership_plans (
    plan_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    plan_name VARCHAR(100) NOT NULL,
    duration_months INTEGER NOT NULL CHECK (duration_months > 0),
    price NUMERIC(12, 2) NOT NULL,
    features_json JSONB,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 25. Bảng Hội Viên Gia Sư Đang Hoạt Động (Tutor Memberships)
DROP TABLE IF EXISTS tutor_memberships CASCADE;
CREATE TABLE tutor_memberships (
    membership_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tutor_id UUID NOT NULL REFERENCES tutors(tutor_id) ON DELETE CASCADE ON UPDATE CASCADE,
    plan_id UUID NOT NULL REFERENCES membership_plans(plan_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    status tutor_membership_state_enum NOT NULL DEFAULT 'ACTIVE',
    amount NUMERIC(12, 2),
    transaction_ref VARCHAR(100),
    payment_status VARCHAR(30) DEFAULT 'PENDING',
    payment_gateway VARCHAR(50) DEFAULT 'VNPay',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_membership_tutor ON tutor_memberships(tutor_id, status);

-- 26. Bảng Giao Dịch Thanh Toán (Payment Transactions)
DROP TABLE IF EXISTS payment_transactions CASCADE;
CREATE TABLE payment_transactions (
    transaction_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    plan_id UUID REFERENCES membership_plans(plan_id) ON DELETE SET NULL ON UPDATE CASCADE,
    booking_id UUID REFERENCES trial_bookings(booking_id) ON DELETE SET NULL ON UPDATE CASCADE,
    amount NUMERIC(12, 2) NOT NULL,
    payment_gateway VARCHAR(50) NOT NULL DEFAULT 'VNPay',
    gateway_transaction_id VARCHAR(100),
    status payment_transaction_status_enum NOT NULL DEFAULT 'PENDING',
    transaction_time TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_payment_user ON payment_transactions(user_id);
CREATE INDEX idx_payment_status ON payment_transactions(status);
CREATE INDEX idx_payment_gateway_ref ON payment_transactions(gateway_transaction_id);

-- 27. Bảng Nhật Ký Xử Lý Cổng Thanh Toán (Payment Processing Logs)
DROP TABLE IF EXISTS payment_processing_logs CASCADE;
CREATE TABLE payment_processing_logs (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    transaction_id UUID NOT NULL REFERENCES payment_transactions(transaction_id) ON DELETE CASCADE ON UPDATE CASCADE,
    request_payload JSONB,
    response_payload JSONB,
    http_status_code INTEGER,
    logged_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ====================================================================================================
-- PHÂN HỆ 6: CHĂM SÓC KHÁCH HÀNG & THÔNG BÁO (SUPPORT & NOTIFICATIONS)
-- ====================================================================================================

-- 28. Bảng Vé Hỗ Trợ Khách Hàng (Support Tickets)
DROP TABLE IF EXISTS support_tickets CASCADE;
CREATE TABLE support_tickets (
    ticket_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    assigned_to UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    escalated_to UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    is_escalated BOOLEAN NOT NULL DEFAULT FALSE,
    escalation_notes TEXT,
    category support_ticket_category_enum NOT NULL,
    subject VARCHAR(200) NOT NULL,
    priority support_ticket_priority_enum NOT NULL DEFAULT 'MEDIUM',
    status support_ticket_status_enum NOT NULL DEFAULT 'OPEN',
    related_transaction_id UUID REFERENCES payment_transactions(transaction_id) ON DELETE SET NULL,
    related_booking_id UUID REFERENCES trial_bookings(booking_id) ON DELETE SET NULL,
    related_membership_id UUID REFERENCES tutor_memberships(membership_id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_tickets_user ON support_tickets(user_id);
CREATE INDEX idx_tickets_assigned ON support_tickets(assigned_to);
CREATE INDEX idx_tickets_escalated ON support_tickets(escalated_to);
CREATE INDEX idx_tickets_status ON support_tickets(status);
CREATE TRIGGER trg_tickets_updated_at BEFORE UPDATE ON support_tickets FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 29. Bảng Tin Nhắn Trong Vé Hỗ Trợ (Ticket Messages)
DROP TABLE IF EXISTS ticket_messages CASCADE;
CREATE TABLE ticket_messages (
    message_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ticket_id UUID NOT NULL REFERENCES support_tickets(ticket_id) ON DELETE CASCADE ON UPDATE CASCADE,
    sender_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    message_text TEXT NOT NULL,
    attachment_url VARCHAR(500),
    sent_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 30. Bảng Thông Báo Hệ Thống (Notifications)
DROP TABLE IF EXISTS notifications CASCADE;
CREATE TABLE notifications (
    notification_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(150) NOT NULL,
    content TEXT NOT NULL,
    notification_type notification_type_enum NOT NULL DEFAULT 'SYSTEM',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 31. Bảng Thông Báo Của Người Dùng (User Notifications)
DROP TABLE IF EXISTS user_notifications CASCADE;
CREATE TABLE user_notifications (
    user_notification_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    notification_id UUID NOT NULL REFERENCES notifications(notification_id) ON DELETE CASCADE ON UPDATE CASCADE,
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    is_read BOOLEAN NOT NULL DEFAULT FALSE,
    read_at TIMESTAMPTZ DEFAULT NULL
);
CREATE INDEX idx_user_notif_unread ON user_notifications(user_id, is_read);

-- PHÂN HỆ 7: CẤU HÌNH HỆ THỐNG, HẠN NGẠCH AI & NHẬT KÝ AUDIT

-- 32. Bảng Cấu Hình Toàn Cục Hệ Thống (System Configurations)
DROP TABLE IF EXISTS system_configurations CASCADE;
CREATE TABLE system_configurations (
    config_key VARCHAR(100) PRIMARY KEY,
    config_value TEXT NOT NULL,
    data_type VARCHAR(20) NOT NULL DEFAULT 'STRING',
    description VARCHAR(255),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE TRIGGER trg_config_updated_at BEFORE UPDATE ON system_configurations FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 33. Bảng Hạn Ngạch Sử Dụng Tính Năng AI Chấm Nét (AI Feature Quotas)
DROP TABLE IF EXISTS ai_feature_quotas CASCADE;
CREATE TABLE ai_feature_quotas (
    quota_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL UNIQUE REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
    daily_ai_drawing_limit INTEGER NOT NULL DEFAULT 50 CHECK (daily_ai_drawing_limit >= 0),
    used_today INTEGER NOT NULL DEFAULT 0 CHECK (used_today >= 0),
    reset_date DATE NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE TRIGGER trg_quotas_updated_at BEFORE UPDATE ON ai_feature_quotas FOR EACH ROW EXECUTE FUNCTION update_timestamp_column();

-- 34. Bảng Nhật Ký Kiểm Toán Hệ Thống (System Audit Logs)
DROP TABLE IF EXISTS system_audit_logs CASCADE;
CREATE TABLE system_audit_logs (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    actor_user_id UUID REFERENCES users(user_id) ON DELETE SET NULL ON UPDATE CASCADE,
    action_type VARCHAR(100) NOT NULL,
    target_entity VARCHAR(100),
    target_id UUID,
    ip_address VARCHAR(45),
    details JSONB,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_audit_actor ON system_audit_logs(actor_user_id);
CREATE INDEX idx_audit_action ON system_audit_logs(action_type);
CREATE INDEX idx_audit_time ON system_audit_logs(created_at);
